cmake_minimum_required(VERSION 3.20)
project(PrismRkLocalExamples VERSION 1.1.0 LANGUAGES CXX)
include(CTest)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/PrismRkLocalSdk.cmake")
foreach(_example capture gnss_status)
  string(REPLACE "_" "-" _name "${_example}")
  add_executable(prism-rklocal-${_name} "../examples/rklocal_${_example}.cpp")
  target_link_libraries(prism-rklocal-${_name} PRIVATE Prism::RkLocal)
  target_compile_options(prism-rklocal-${_name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  set_target_properties(prism-rklocal-${_name} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  if(BUILD_TESTING)
    add_test(NAME prism-rklocal-${_name}-help COMMAND prism-rklocal-${_name} --help)
  endif()
endforeach()

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
