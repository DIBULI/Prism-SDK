cmake_minimum_required(VERSION 3.20)
project(PrismSdkExamples VERSION 1.2.1 LANGUAGES C CXX)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" processor)
if(WIN32 AND processor MATCHES "^(amd64|x86_64|x64)$")
  set(platform windows-x64)
elseif(APPLE AND processor MATCHES "^(arm64|aarch64)$")
  set(platform macos-arm64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND processor MATCHES "^(amd64|x86_64|x64)$")
  set(platform linux-x64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND processor MATCHES "^(arm64|aarch64)$")
  set(platform linux-arm64)
else()
  message(FATAL_ERROR "Supported platforms: Windows x64, macOS arm64, Linux x64/arm64")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/runtime/ros/${platform}")
find_package(PrismUsbSdk 1.2.1 EXACT CONFIG REQUIRED)
file(GLOB examples CONFIGURE_DEPENDS examples/*.cpp)
foreach(source IN LISTS examples)
  get_filename_component(name "${source}" NAME_WE)
  add_executable(prism-${name} "${source}")
  target_link_libraries(prism-${name} PRIVATE Prism::UsbSdk)
endforeach()
