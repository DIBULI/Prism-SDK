cmake_minimum_required(VERSION 3.20)
project(PrismRkLocalExamples VERSION 1.1.0 LANGUAGES C)
include(CTest)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/PrismRkLocalSdk.cmake")
foreach(example capture gnss_status)
  string(REPLACE "_" "-" name "${example}")
  add_executable(prism-rklocal-${name} "../rklocal_${example}.c")
  target_link_libraries(prism-rklocal-${name} PRIVATE Prism::RkLocal)
  target_compile_features(prism-rklocal-${name} PRIVATE c_std_11)
  target_compile_options(prism-rklocal-${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${name}-help COMMAND prism-rklocal-${name} --help)
endforeach()
